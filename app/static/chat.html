function speakText(text) { if (!('speechSynthesis' in window)) return; window.speechSynthesis.cancel(); // تنظيف النص تماماً من الشرطات، النجوم، والرموز المزعجة قبل القراءة let cleanText = text .replace(/[_#*`~>-]/g, ' ') // إزالة الشرطات السفلية والنجوم والرموز البرمجية .replace(/\$\$(.*?)\$\$/g, ' نقطة ') .replace(/\$(.*?)\$/, ' ') .replace(/OM/g, 'أو إم') .replace(/O'M/g, 'أو فتحه إم') .replace(/OO'/g, 'أو أو فتحه'); // ترتيب الجمل لنطقها بسلاسة let sentences = cleanText.split(/[.،؟\n]/); let shortText = sentences.filter(s => s.trim().length > 0).slice(0, 5).join('. '); const utterance = new SpeechSynthesisUtterance(shortText); utterance.lang = 'ar-SA'; // أو ar-LB حسب توفرها في متصفح الهاتف utterance.rate = 1.0; window.speechSynthesis.speak(utterance); }
